"use client";
import React, { useState, useRef } from "react";

// --- Timezone Clock Util ---
const TIME_ZONES = [
  { label: "UTC", zone: "UTC" },
  { label: "WIB (Jakarta)", zone: "Asia/Jakarta" },
  { label: "London", zone: "Europe/London" },
  { label: "Tokyo", zone: "Asia/Tokyo" },
  { label: "US/Pacific", zone: "America/Los_Angeles" },
];
function useClocks() {
  const [now, setNow] = useState(Date.now());
  React.useEffect(() => {
    const t = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(t);
  }, []);
  return TIME_ZONES.map(({ label, zone }) => ({
    label,
    time: new Intl.DateTimeFormat("en-GB", {
      hour: "2-digit",
      minute: "2-digit",
      second: "2-digit",
      hour12: false,
      timeZone: zone,
    }).format(new Date(now)),
  }));
}

// --- Main Page: UI & Logic ---
interface Job {
  id: string;
  prompt: string;
  style: string;
  status: "processing" | "done" | "error";
  videoUrl?: string;
}

const styleOptions = ["Cinematic", "Anime", "Realistic", "Cartoon", "Vlog"];

export default function Page() {
  const clocks = useClocks();
  const formRef = useRef<HTMLFormElement>(null);
  const [prompt, setPrompt] = useState("");
  const [style, setStyle] = useState(styleOptions[0]);
  const [jobs, setJobs] = useState<Job[]>([]);
  const [loading, setLoading] = useState(false);

  // Simulasi backend (bisa diganti fetch ke "/api/generate-video")
  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    if (!prompt.trim()) return;
    setLoading(true);
    const jobId = "job-" + Math.random().toString(36).slice(2, 9);
    // Tambah ke job list (status processing)
    setJobs((prev) => [
      { id: jobId, prompt, style, status: "processing" },
      ...prev,
    ]);
    setPrompt("");
    // Simulasi job selesai setelah 4s
    setTimeout(() => {
      setJobs((prev) =>
        prev.map((job) =>
          job.id === jobId
            ? {
                ...job,
                status: "done",
                videoUrl: "https://www.w3schools.com/html/mov_bbb.mp4",
              }
            : job
        )
      );
      setLoading(false);
    }, 4000);
    formRef.current?.reset();
  }

  return (
    <main className="min-h-screen bg-slate-950 text-white flex flex-col items-center p-4">
      <header className="w-full max-w-3xl text-center mb-8">
        <h1 className="text-4xl font-bold mb-2 tracking-tight">AIGen-VISION SUPREME</h1>
        <p className="mb-3 text-slate-300">AI Text-to-Video Generator <b>(Demo)</b></p>
        <section className="mb-4 flex justify-center gap-5 flex-wrap text-xs md:text-sm">
          {clocks.map((c) => (
            <div key={c.label} className="bg-slate-800 px-2 py-1 rounded shadow">
              <span className="font-semibold">{c.label}</span>: <span className="tabular-nums">{c.time}</span>
            </div>
          ))}
        </section>
      </header>
      {/* PROMPT INPUT */}
      <form
        ref={formRef}
        onSubmit={handleSubmit}
        className="bg-slate-800 rounded-lg flex flex-col md:flex-row gap-3 w-full max-w-xl items-end mb-8 p-4 shadow-lg border border-slate-700"
      >
        <input
          type="text"
          className="text-black px-3 py-2 rounded border w-full flex-1"
          placeholder="Describe your video prompt (e.g. spaceship flies over city, sunset color grading)..."
          required
          disabled={loading}
          value={prompt}
          minLength={5}
          maxLength={160}
          onChange={(e) => setPrompt(e.target.value)}
        />
        <select
          className="px-3 py-2 rounded border bg-slate-900 text-white"
          value={style}
          disabled={loading}
          onChange={(e) => setStyle(e.target.value)}
        >
          {styleOptions.map((s) => (
            <option key={s} value={s}>
              {s}
            </option>
          ))}
        </select>
        <button
          className="bg-emerald-500 hover:bg-emerald-600 px-6 py-2 rounded text-white font-semibold tracking-wider flex-shrink-0"
          disabled={loading}
          type="submit"
        >
          {loading ? "Processing..." : "Generate"}
        </button>
      </form>
      {/* JOB LIST */}
      <section className="w-full max-w-xl">
        <h2 className="text-lg font-semibold mb-3">Your Video Jobs</h2>
        <ul className="flex flex-col gap-3">
          {jobs.length === 0 && (
            <li className="text-center text-slate-400 italic">No jobs yet. Submit a prompt!</li>
          )}
          {jobs.map((job) => (
            <li
              key={job.id}
              className="bg-slate-800 border border-slate-700 rounded-lg p-4 flex flex-col gap-2 shadow"
            >
              <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-1">
                <div>
                  <span className="font-bold">{job.style}</span>
                  <span className="mx-2">&bull;</span>
                  <span className="text-slate-300">{job.prompt}</span>
                </div>
                <span
                  className={
                    "inline-block min-w-[90px] text-xs text-center px-2 py-1 rounded font-semibold " +
                    (job.status === "done"
                      ? "bg-emerald-700"
                      : job.status === "processing"
                      ? "bg-yellow-700"
                      : "bg-red-700")
                  }
                >
                  {job.status === "done"
                    ? "Ready"
                    : job.status === "processing"
                    ? "Processing..."
                    : "Error"}
                </span>
              </div>
              {job.status === "done" && job.videoUrl && (
                <video
                  src={job.videoUrl}
                  controls
                  className="rounded mt-2 w-full max-h-60 border"
                />
              )}
            </li>
          ))}
        </ul>
      </section>
      {/* FOOTER */}
      <footer className="w-full max-w-xl text-xs text-slate-600 text-center py-8 mt-10 border-t border-slate-800">
        &copy; 2025 AIGen-VISION SUPREME &mdash; Text-to-Video AI <br />
        <span className="text-slate-500">This is a demo. No real AI inference — yet!</span>
      </footer>
    </main>
  );
}
